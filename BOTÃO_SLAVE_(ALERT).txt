#include <SoftwareSerial.h>

// Bluetooth
SoftwareSerial BT(3, 2);

#define BT_STATE 6
#define LED_CONEXAO 10

// LEDs
#define led1 12
#define led2 11
#define led3 9
#define led4 8

#define buzzer 7

// Sensor ultrassônico
#define TRIG 5
#define ECHO 4

// Distância para ativar o alerta
#define DISTANCIA_ALERTA 20

// Guarda o estado anterior do Bluetooth
bool conectadoAnterior = false;

long medirDistancia() {

  // Garante que o TRIG começa desligado
  digitalWrite(TRIG, LOW);
  delayMicroseconds(2);

  // Envia pulso de 10 microssegundos
  digitalWrite(TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG, LOW);

  // Mede o tempo do retorno do ECHO
  long duracao = pulseIn(ECHO, HIGH, 30000);

  // Se não recebeu retorno
  if (duracao == 0) {
    return 999;
  }

  // Converte tempo em distância em centímetros
  long distancia = duracao * 0.0343 / 2;

  return distancia;
}

void executarAlerta() {

  for (int i = 0; i < 2; i++) {

    tone(buzzer, 500);

    digitalWrite(led1, HIGH);
    digitalWrite(led3, HIGH);

    digitalWrite(led2, LOW);
    digitalWrite(led4, LOW);

    delay(500);

    noTone(buzzer);

    digitalWrite(led1, LOW);
    digitalWrite(led3, LOW);

    digitalWrite(led2, HIGH);
    digitalWrite(led4, HIGH);

    delay(500);
  }

  noTone(buzzer);

  digitalWrite(led1, LOW);
  digitalWrite(led2, LOW);
  digitalWrite(led3, LOW);
  digitalWrite(led4, LOW);
}

void setup() {

  Serial.begin(9600);

  // Bluetooth
  BT.begin(9600);

  // LEDs
  pinMode(led1, OUTPUT);
  pinMode(led2, OUTPUT);
  pinMode(led3, OUTPUT);
  pinMode(led4, OUTPUT);

  pinMode(buzzer, OUTPUT);

  pinMode(BT_STATE, INPUT);

  pinMode(LED_CONEXAO, OUTPUT);

  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);

  digitalWrite(led1, LOW);
  digitalWrite(led2, LOW);
  digitalWrite(led3, LOW);
  digitalWrite(led4, LOW);

  digitalWrite(LED_CONEXAO, LOW);

  digitalWrite(TRIG, LOW);

  noTone(buzzer);

  Serial.println("Sistema iniciado!");
}

void loop() {

  // Bluetooth
  bool conectado = digitalRead(BT_STATE);

  digitalWrite(LED_CONEXAO, conectado);

  if (conectado != conectadoAnterior) {

    if (conectado) {
      Serial.println("Bluetooth CONECTADO!");
    } 
    else {
      Serial.println("Bluetooth DESCONECTADO!");
    }

    conectadoAnterior = conectado;
  }

  if (BT.available()) {

    char c = BT.read();

    Serial.print("Recebido: ");
    Serial.println(c);

    if (c == 'A') {

      // Responde K pelo Bluetooth
      BT.write('K');

      executarAlerta();
    }
  }

  long distancia = medirDistancia();

  Serial.print("Distancia: ");
  Serial.print(distancia);
  Serial.println(" cm");

  if (distancia <= DISTANCIA_ALERTA) {

    Serial.println("!!! OBJETO DETECTADO !!!");

    if (conectado) {
      BT.println("ALERTA: OBJETO DETECTADO!");
    }

    executarAlerta();
  }

  delay(100);
}
