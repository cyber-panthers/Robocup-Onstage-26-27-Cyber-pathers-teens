#include <SoftwareSerial.h>
#include <Adafruit_NeoPixel.h>

// =====================================================
// BLUETOOTH HC-05
// =====================================================

SoftwareSerial BT(3, 2);

// HC-05
#define BT_STATE 8
#define LED_CONEXAO 13


// =====================================================
// FITA LED
// =====================================================

#define LED_PIN 5
#define NUM_LEDS 87

Adafruit_NeoPixel fita(NUM_LEDS, LED_PIN, NEO_GRB + NEO_KHZ800);


// =====================================================
// SENSOR ULTRASSÔNICO 1
// =====================================================

const int trigPin1 = 7;
const int echoPin1 = 6;

// Distância para ativar o sensor 1
const float LIMITE_SENSOR1 = 10.0;


// =====================================================
// SENSOR ULTRASSÔNICO 2
// =====================================================

const int trigPin2 = 11;
const int echoPin2 = 10;

// Distância para ativar o sensor 2
const float LIMITE_SENSOR2 = 5.0;


// =====================================================
// CONTROLE
// =====================================================

bool enviado = false;


// =====================================================
// MEDIR DISTÂNCIA
// =====================================================

float distancia(int trigPin, int echoPin) {

  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);

  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);

  digitalWrite(trigPin, LOW);

  long duracao = pulseIn(echoPin, HIGH, 30000);

  // Sem retorno do sensor
  if (duracao == 0) {
    return 999;
  }

  float distanciaCM = duracao * 0.0343 / 2;

  return distanciaCM;
}


// =====================================================
// ACENDER LED POR LED
// =====================================================

void acenderLedPorLed() {

  Serial.println();
  Serial.println("Acendendo fita LED por LED...");

  // Apaga tudo primeiro
  fita.clear();
  fita.show();

  // Acende um LED por vez
  for (int i = 0; i < NUM_LEDS; i++) {

    fita.setPixelColor(i, 255, 0, 0);

    fita.show();

    // Velocidade da animação
    delay(15);
  }

  Serial.println("Fita totalmente acesa!");
}


// =====================================================
// ENVIAR SINAL PARA O SLAVE
// =====================================================

void enviarSinal() {

  Serial.println("Enviando sinal A...");

  // Envia 3 vezes para aumentar
  // a chance de recebimento
  for (int i = 0; i < 3; i++) {

    BT.write('A');

    delay(50);
  }

  Serial.println("Sinal A enviado!");
}


// =====================================================
// PISCAR FITA
// =====================================================

void piscarFita() {

  Serial.println("Piscando fita...");

  unsigned long inicio = millis();

  while (millis() - inicio < 3300) {

    // =================================================
    // AZUL
    // =================================================

    for (int i = 0; i < NUM_LEDS; i++) {

      fita.setPixelColor(
        i,
        fita.Color(0, 0, 255)
      );
    }

    fita.show();

    delay(500);


    // =================================================
    // VERMELHO
    // =================================================

    for (int i = 0; i < NUM_LEDS; i++) {

      fita.setPixelColor(
        i,
        fita.Color(255, 0, 0)
      );
    }

    fita.show();

    delay(500);
  }


  // =================================================
  // APAGA
  // =================================================

  fita.clear();
  fita.show();

  Serial.println("Fita apagada.");
}


// =====================================================
// EXECUTAR SEQUÊNCIA
// =====================================================

void executarSequencia() {

  Serial.println();
  Serial.println("==============================");
  Serial.println("SEQUENCIA INICIADA");
  Serial.println("==============================");

  // Acende os LEDs
  acenderLedPorLed();

  // Envia sinal pelo Bluetooth
  enviarSinal();

  // Pisca azul e vermelho
  piscarFita();

  Serial.println();
  Serial.println("SEQUENCIA FINALIZADA");
  Serial.println("==============================");
}


// =====================================================
// SETUP
// =====================================================

void setup() {

  // =================================================
  // SERIAL MONITOR
  // =================================================

  Serial.begin(9600);


  // =================================================
  // BLUETOOTH
  // =================================================

  BT.begin(9600);


  // =================================================
  // SENSOR 1
  // TRIG = 7
  // ECHO = 6
  // =================================================

  pinMode(trigPin1, OUTPUT);
  pinMode(echoPin1, INPUT);


  // =================================================
  // SENSOR 2
  // TRIG = 11
  // ECHO = 10
  // =================================================

  pinMode(trigPin2, OUTPUT);
  pinMode(echoPin2, INPUT);


  // =================================================
  // BLUETOOTH
  // =================================================

  pinMode(BT_STATE, INPUT);
  pinMode(LED_CONEXAO, OUTPUT);


  // =================================================
  // FITA LED
  // =================================================

  fita.begin();

  fita.clear();

  fita.show();


  // =================================================
  // MENSAGENS INICIAIS
  // =================================================

  Serial.println();
  Serial.println("==============================");
  Serial.println("MASTER INICIADO");
  Serial.println("==============================");

  Serial.println("Bluetooth aguardando...");

  Serial.println();
  Serial.println("Sensor 1:");
  Serial.println("TRIG = 7");
  Serial.println("ECHO = 6");
  Serial.println("Ativacao = 10 cm");

  Serial.println();
  Serial.println("Sensor 2:");
  Serial.println("TRIG = 11");
  Serial.println("ECHO = 10");
  Serial.println("Ativacao = 5 cm");

  Serial.println();
}


// =====================================================
// LOOP
// =====================================================

void loop() {

  // =================================================
  // BLUETOOTH
  // =================================================

  bool conectado = digitalRead(BT_STATE);

  digitalWrite(
    LED_CONEXAO,
    conectado
  );


  // =================================================
  // DISTÂNCIA SENSOR 1
  // =================================================

  float dist1 = distancia(
    trigPin1,
    echoPin1
  );


  // Pequena pausa para evitar interferência
  // entre os dois sensores
  delay(5);


  // =================================================
  // DISTÂNCIA SENSOR 2
  // =================================================

  float dist2 = distancia(
    trigPin2,
    echoPin2
  );


  // =================================================
  // MOSTRAR SENSOR 1
  // =================================================

  Serial.print("Sensor 1: ");

  if (dist1 >= 999) {

    Serial.print("SEM LEITURA");

  } else {

    Serial.print(dist1, 1);
    Serial.print(" cm");
  }


  // =================================================
  // MOSTRAR SENSOR 2
  // =================================================

  Serial.print(" | Sensor 2: ");

  if (dist2 >= 999) {

    Serial.print("SEM LEITURA");

  } else {

    Serial.print(dist2, 1);
    Serial.print(" cm");
  }


  // =================================================
  // MOSTRAR BLUETOOTH
  // =================================================

  Serial.print(" | Bluetooth: ");

  if (conectado) {

    Serial.println("CONECTADO");

  } else {

    Serial.println("DESCONECTADO");
  }


  // =================================================
  // DETECÇÃO
  // =================================================
  //
  // SENSOR 1:
  // ativa quando <= 10 cm
  //
  // SENSOR 2:
  // ativa quando <= 5 cm
  //
  // QUALQUER UM DOS DOIS PODE ATIVAR
  // =================================================

  if (
    (dist1 <= LIMITE_SENSOR1 ||
     dist2 <= LIMITE_SENSOR2)
    &&
    !enviado
  ) {

    enviado = true;

    Serial.println();
    Serial.println(">>> OBJETO DETECTADO! <<<");


    // =================================================
    // INFORMAR QUAL SENSOR DETECTOU
    // =================================================

    if (dist1 <= LIMITE_SENSOR1) {

      Serial.println(
        ">>> SENSOR 1 ATIVOU! (<= 10 cm)"
      );
    }


    if (dist2 <= LIMITE_SENSOR2) {

      Serial.println(
        ">>> SENSOR 2 ATIVOU! (<= 5 cm)"
      );
    }


    // =================================================
    // EXECUTAR SEQUÊNCIA
    // =================================================

    executarSequencia();
  }


  // =================================================
  // RESET DA DETECÇÃO
  // =================================================
  //
  // Para permitir uma nova ativação, o objeto precisa
  // sair da área dos dois sensores.
  //
  // Sensor 1 precisa passar de 15 cm.
  // Sensor 2 precisa passar de 10 cm.
  // =================================================

  if (
    dist1 > 15 &&
    dist2 > 10
  ) {

    enviado = false;
  }


  // =================================================
  // ATUALIZAÇÃO
  // =================================================

  delay(100);
}
